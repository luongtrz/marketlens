create table public.daily_factor_snapshots (
  id bigint generated always as identity not null,
  symbol text not null,
  snapshot_date date not null,
  factors_json jsonb not null,
  updated_at timestamp with time zone null default now(),
  news_count integer null,
  article_ids text[] null,
  taxonomy_events jsonb null default '[]'::jsonb,
  group_vector smallint[] null default '{}'::smallint[],
  type_vector smallint[] null default '{}'::smallint[],
  factor_ledge_json jsonb null default '{}'::jsonb,
  factor_vector smallint[] null default '{}'::smallint[],
  generation_meta jsonb null default '{}'::jsonb,
  constraint daily_factor_snapshots_pkey primary key (id),
  constraint daily_factor_snapshots_symbol_snapshot_date_key unique (symbol, snapshot_date),
  constraint daily_factor_snapshots_factor_vector_len_chk check ((cardinality(factor_vector) = any (array[0, 75]))),
  constraint daily_factor_snapshots_group_vector_len_chk check ((cardinality(group_vector) = any (array[0, 13]))),
  constraint daily_factor_snapshots_type_vector_len_chk check ((cardinality(type_vector) = any (array[0, 62])))
) TABLESPACE pg_default;

create index IF not exists idx_daily_factor_snapshots_symbol_snapshot_date_desc on public.daily_factor_snapshots using btree (symbol, snapshot_date desc) TABLESPACE pg_default;

create index IF not exists idx_daily_factor_snapshots_factor_ledge_json_gin on public.daily_factor_snapshots using gin (factor_ledge_json) TABLESPACE pg_default;

create table public.news_articles (
  id bigint generated by default as identity not null,
  header text not null,
  content text null,
  publish_at timestamp with time zone null,
  crawled_at timestamp with time zone null default now(),
  source_url text null,
  sentiment_score numeric null,
  summary text null,
  constraint news_articles_pkey primary key (id),
  constraint news_articles_source_url_key unique (source_url),
  constraint news_articles_sentiment_score_check check (
    (
      (sentiment_score >= '-1.00'::numeric)
      and (sentiment_score <= 1.00)
    )
  )
) TABLESPACE pg_default;
